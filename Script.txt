local function ObtenerMejorTarget()
	local mejorTarget = nil
	local menorDistancia = math.huge
	local centroPantallaX = Camera.ViewportSize.X / 2
	local centroPantallaY = Camera.ViewportSize.Y / 2

	for _, jugador in pairs(Players:GetPlayers()) do
		if jugador == LP then continue end
		if not jugador.Character then continue end

		-- Team Check
		if S.TeamCheck and jugador.Team and LP.Team and jugador.Team == LP.Team then continue end

		local Humanoide = jugador.Character:FindFirstChild("Humanoid")
		if not Humanoide or Humanoide.Health <= 0 then continue end

		-- Buscar parte del cuerpo (con respaldo en cadena)
		local ParteObjetivo = 
			jugador.Character:FindFirstChild(S.TargetPart) or
			jugador.Character:FindFirstChild("Head") or
			jugador.Character:FindFirstChild("UpperTorso") or
			jugador.Character:FindFirstChild("HumanoidRootPart")
		if not ParteObjetivo then continue end

		-- Verificar si está en pantalla
		local PosicionEnPantalla, Visible = Camera:WorldToViewportPoint(ParteObjetivo.Position)
		if not Visible then continue end

		-- Calcular distancia al centro
		local DistanciaAlCentro = math.sqrt(
			(PosicionEnPantalla.X - centroPantallaX)^2 + 
			(PosicionEnPantalla.Y - centroPantallaY)^2
		)

		if DistanciaAlCentro < menorDistancia then
			menorDistancia = DistanciaAlCentro
			mejorTarget = ParteObjetivo
		end
	end

	return mejorTarget
end

-- 🔒 HOOK INDESTRUCTIBLE — se repara solo si se rompe
local function RestaurarHook()
	if HookActivo then return end
	pcall(function()
		local mt = getrawmetatable(Mouse)
		local oldIndex = mt.__index
		setreadonly(mt, false)
		mt.__index = newcclosure(function(self, clave)
			if S.SilentAim and S.CurrentTarget then
				if clave == "Hit" then
					return CFrame.new(S.CurrentTarget.Position)
				end
				if clave == "Target" then
					return S.CurrentTarget
				end
			end
			return oldIndex(self, clave)
		end)
		setreadonly(mt, true)
		HookActivo = true
	end)
end

-- Recuperar TODO al respawnear
LP.CharacterAdded:Connect(function()
	task.wait(0.3)
	Mouse = LP:GetMouse()
	HookActivo = false
	RestaurarHook()
end)

-- Bucle principal
local function BucleAim()
	if not S.SilentAim then
		S.CurrentTarget = nil
		return
	end
	RestaurarHook() -- Repara el hook si se rompió
	S.CurrentTarget = ObtenerMejorTarget()
end

-- ==============================================
-- 🌈 CHAMS — PERSISTENTES SIEMPRE
-- ==============================================
local function CrearChams(Jugador)
	if not Jugador.Character then return end
	if ChamsCreados[Jugador] then
		pcall(function() ChamsCreados[Jugador]:Destroy() end)
	end
	local Resaltado = Instance.new("Highlight")
	Resaltado.Name = "ZZXP_Chams"
	Resaltado.FillTransparency = 0.3
	Resaltado.OutlineTransparency = 0
	Resaltado.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	Resaltado.Parent = Jugador.Character
	ChamsCreados[Jugador] = Resaltado
end

-- Recrear automáticamente al respawnear
for _, Jugador in pairs(Players:GetPlayers()) do
	Jugador.CharacterAdded:Connect(function()
		task.wait(0.8)
		if S.Chams then CrearChams(Jugador) end
	end)
end
Players.PlayerAdded:Connect(function(Jugador)
	Jugador.CharacterAdded:Connect(function()
		task.wait(0.8)
		if S.Chams then CrearChams(Jugador) end
	end)
end)

local function AplicarChamsTodos()
	for _, Jugador in pairs(Players:GetPlayers()) do
		if Jugador ~= LP and Jugador.Character then
			CrearChams(Jugador)
		end
	end
end

local function QuitarTodoChams()
	for _, Resaltado in pairs(ChamsCreados) do
		pcall(function() Resaltado:Destroy() end)
	end
	table.clear(ChamsCreados)
end

local function BucleChams()
	if not S.Chams then return end
	local ColorActual
	if S.RainbowChams then
		local t = tick() * 2
		ColorActual = Color3.new(
			math.sin(t)*0.5 + 0.5,
			math.sin(t+2)*0.5 + 0.5,
			math.sin(t+4)*0.5 + 0.5
		)
	else
		ColorActual = S.ColorChams
	end
	for Jugador, Resaltado in pairs(ChamsCreados) do
		if not Resaltado or not Resaltado.Parent then
			CrearChams(Jugador)
			Resaltado = ChamsCreados[Jugador]
		end
		if Resaltado then
			Resaltado.FillColor = ColorActual
			Resaltado.OutlineColor = ColorActual
		end
	end
end

Players.PlayerRemoving:Connect(function(Jugador)
	ChamsCreados[Jugador] = nil
end)
